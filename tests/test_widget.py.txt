"""
Тесты для модуля widget.
"""

import pytest
from src.widget import mask_account_card, get_date


@pytest.mark.parametrize(
    "input_str, expected",
    [
        ("Visa 1234567812345678", "Visa 1234 56** **** 5678"),
        ("MasterCard 1111222233334444", "MasterCard 1111 22** **** 4444"),
        ("Счет 12345678901234567890", "Счет **7890"),
        ("Счет 98765432109876543210", "Счет **3210"),
        ("Visa", "Visa"),  # без номера
    ],
)
def test_mask_account_card(input_str, expected):
    """Тест маскирования карт и счетов."""
    assert mask_account_card(input_str) == expected


@pytest.mark.parametrize(
    "date_str, expected",
    [
        ("2024-03-15T10:30:00", "15.03.2024"),
        ("2024-12-01T00:00:00", "01.12.2024"),
        ("2023-01-01", "01.01.2023"),
        ("не дата", "не дата"),  # если функция возвращает как есть
    ],
)
def test_get_date(date_str, expected):
    """Тест преобразования даты."""
    assert get_date(date_str) == expected
